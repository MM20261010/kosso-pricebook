cmake_minimum_required(VERSION 3.22)
project(local_paddle LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep debug UI/symbols, but compare optimized math rather than -O0 native code.
set(CMAKE_C_FLAGS_DEBUG "-g -O2")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O2")
set(NCNN_VERSION 20261020)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native-deps")
if(NOT EXISTS "${DEPS}/ncnn/CMakeLists.txt" OR NOT EXISTS "${DEPS}/opencv/CMakeLists.txt")
 message(FATAL_ERROR "Run python3 scripts/prepare-native.py before building")
endif()
foreach(flag NCNN_INT8 NCNN_BF16 NCNN_ARM82 NCNN_ARM82DOT NCNN_ARM82FP16FML NCNN_ARM84BF16 NCNN_ARM84I8MM NCNN_ARM86SVE NCNN_AVXVNNI NCNN_AVX512)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(NCNN_DISABLE_RTTI OFF CACHE BOOL "" FORCE)
set(NCNN_VALIDATION ON CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_THREADS OFF CACHE BOOL "" FORCE)
set(NCNN_VULKAN OFF CACHE BOOL "" FORCE)
set(NCNN_STDIO OFF CACHE BOOL "" FORCE)
set(NCNN_STRING ON CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/ncnn" "${CMAKE_BINARY_DIR}/ncnn")
set(BUILD_LIST core,imgproc CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_DOCS BUILD_ANDROID_PROJECTS BUILD_ANDROID_EXAMPLES WITH_ADE WITH_ZLIB WITH_LAPACK WITH_IPP WITH_ITT WITH_OPENCL WITH_FFMPEG WITH_GSTREAMER WITH_PROTOBUF WITH_VULKAN WITH_CUDA WITH_OPENEXR WITH_JASPER WITH_WEBP WITH_JPEG WITH_PNG WITH_TIFF)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(CV_TRACE OFF CACHE BOOL "" FORCE)
set(OPENCV_ENABLE_NONFREE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/opencv" "${CMAKE_BINARY_DIR}/opencv")
add_library(paddle_core STATIC paddle_core.cpp "${DEPS}/clipper.cpp")
target_include_directories(paddle_core PUBLIC . "${DEPS}" "${DEPS}/opencv/modules/core/include" "${DEPS}/opencv/modules/imgproc/include" "${CMAKE_BINARY_DIR}/opencv" "${CMAKE_BINARY_DIR}")
target_compile_features(paddle_core PUBLIC cxx_std_17)
target_compile_options(paddle_core PRIVATE -fexceptions -frtti)
target_link_libraries(paddle_core PUBLIC ncnn opencv_core opencv_imgproc)
if(ANDROID)
 add_library(local_paddle SHARED paddle_jni.cpp)
 target_compile_options(local_paddle PRIVATE -fexceptions -frtti)
 target_link_libraries(local_paddle PRIVATE paddle_core android jnigraphics)
 target_link_options(local_paddle PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,noexecstack" "-Wl,-z,relro,-z,now" "-Wl,--exclude-libs,ALL")
else()
 add_executable(paddle_core_test "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/paddle_core_test.cpp")
 target_link_libraries(paddle_core_test PRIVATE paddle_core)
endif()
